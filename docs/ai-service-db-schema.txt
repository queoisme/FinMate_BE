======================================================================
2. CHI TIET TUNG BANG
======================================================================

--- NHOM 1: DU LIEU HUAN LUYEN ---

raw_samples                      Thong bao tho da an danh
  id                  uuid PK
  source              varchar(32)  not null
  package_name        text         not null
  notification_title  text
  notification_body   text         not null
  received_at         timestamptz  not null
  user_id_hash        varchar(64)             (SHA-256 cua user_id)
  content_hash        varchar(64)  not null   UNIQUE (uq_raw_samples_content_hash)
  created_at          timestamptz
  Index: package_name; created_at

labeled_samples                  Nhan cua moi mau
  id                  uuid PK
  raw_sample_id       uuid FK -> raw_samples.id (CASCADE), UNIQUE
  is_financial        boolean not null
  transaction_type    varchar(16)
  amount_cents        bigint
  merchant_name       text
  category_slug       varchar(32)
  labeled_by          varchar(32) not null
  labeled_at          timestamptz
  is_gold             boolean
  Index: category_slug

sample_splits                    Chia train / val / test
  id                  uuid PK
  labeled_sample_id   uuid FK -> labeled_samples.id (CASCADE), UNIQUE
  split               varchar(16) not null
  split_seed          integer not null
  assigned_at         timestamptz
  Index: split

--- NHOM 2: MODEL VA HUAN LUYEN ---

training_jobs                    Moi lan chay train
  id                  uuid PK
  stage               varchar(32) not null
  status              varchar(16) not null
  params              jsonb
  sample_count        integer
  started_at          timestamptz
  finished_at         timestamptz
  error_message       text
  created_at          timestamptz
  Index: (stage, created_at)

model_versions                   Registry model
  id                  uuid PK
  stage               varchar(32) not null
  version             varchar(32) not null
  artifact_path       text not null
  status              varchar(16) not null
  metrics             jsonb
  training_job_id     uuid FK -> training_jobs.id (SET NULL), nullable
  trained_at          timestamptz
  notes               text
  created_at, updated_at timestamptz
  UNIQUE (stage, version)

--- NHOM 3: DANH GIA MODEL ---

evaluation_runs                  Mot lan danh gia
  id                  uuid PK
  model_version_id    uuid FK -> model_versions.id (CASCADE)
  split               varchar(16) not null
  sample_count        integer not null
  accuracy            float not null
  macro_f1            float not null
  metrics             jsonb
  notes               text
  created_at          timestamptz
  Index: (model_version_id, created_at)

evaluation_category_metrics      Chi so theo tung nhan
  id                  uuid PK
  evaluation_run_id   uuid FK -> evaluation_runs.id (CASCADE)
  label               varchar(64) not null
  precision, recall, f1  float not null
  support             integer not null
  UNIQUE (evaluation_run_id, label)

evaluation_predictions           Du doan tung mau
  id                  uuid PK
  evaluation_run_id   uuid FK -> evaluation_runs.id (CASCADE)
  labeled_sample_id   uuid FK -> labeled_samples.id (CASCADE)
  predicted_label     varchar(64) not null
  true_label          varchar(64) not null
  confidence          float not null
  is_correct          boolean not null
  Index: (evaluation_run_id, is_correct)

--- NHOM 4: VAN HANH (KHI CHAY THAT) ---

pipeline_requests                Nhat ky moi lan phan tich
  id                  uuid PK
  backend_request_id  uuid not null, UNIQUE (uq_pipeline_requests_backend_id)
  user_id_hash        varchar(64) not null
  package_name        text
  received_at         timestamptz
  pipeline_result     varchar(32) not null
  classifier_label    varchar(32)
  classifier_confidence float
  amount_cents        bigint
  transaction_type    varchar(16)
  merchant_name       text
  description         text
  transacted_at       timestamptz
  balance_after_cents bigint
  extraction_confidence float
  extraction_method   varchar(16)
  matched_pattern_name varchar(120)
  category_slug       varchar(32)
  categorization_confidence float
  is_potential_duplicate boolean
  duplicate_of_request_id uuid          (lien ket mem, khong FK)
  classifier_version, extractor_version, categorizer_version  varchar(32)
  raw_sample_id       uuid FK -> raw_samples.id (SET NULL), nullable
  processing_ms       integer
  error_message       text
  created_at          timestamptz
  Index: (user_id_hash, transacted_at); created_at

provider_patterns                Regex theo tung ngan hang
  id                  uuid PK
  provider_key        text
  package_name        text
  pattern_name        text
  regex               text
  transaction_type    varchar(16)
  priority            integer
  sample_text         text
  is_active           boolean
  notes               text
  created_at, updated_at timestamptz
  UNIQUE + index theo package_name (xem migration 0001)

user_feedback                    Nguoi dung sua danh muc
  id                  uuid PK
  backend_transaction_id_hash varchar(64) not null   (SHA-256)
  user_id_hash        varchar(64) not null
  pipeline_request_id uuid                 (lien ket mem, khong FK)
  package_name        text
  predicted_category  varchar(32)
  corrected_category  varchar(32)
  feedback_type       varchar(32) not null
  processed_at        timestamptz
  batch_job_id        uuid FK -> feedback_batch_jobs.id (SET NULL), nullable
  created_at          timestamptz
  Index: pipeline_request_id (va mot index khac, xem migration 0004)

feedback_batch_jobs              Moi lan gom feedback thanh nhan
  id                  uuid PK
  status              varchar(16) not null
  started_at          timestamptz
  finished_at         timestamptz
  feedback_count      integer
  created_labeled_samples integer
  error_message       text
  created_at          timestamptz
